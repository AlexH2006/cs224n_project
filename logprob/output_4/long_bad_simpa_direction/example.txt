=== CONTEXT ===
import Mathlib

theorem long_bad_simpa_direction (n m : Nat) (h : n = m) : m + 0 = n := by

=== PROOF ===
  have h1 : 0 + n = m := by
    simpa using h
  have h2 : n = m := by
    exact h
  have h3 : m + 0 = m := by
    simpa using Nat.add_zero m
  calc
    m + 0 = m := h3
    _ = n := by
      simpa using h2
