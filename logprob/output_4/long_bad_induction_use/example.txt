=== CONTEXT ===
import Mathlib

theorem long_bad_induction_use (n : Nat) : n + 0 = n := by

=== PROOF ===
  induction n with
  | zero =>
      simp
  | succ k hk =>
      have h1 : k + 0 = k := by
        exact rfl
      have h2 : k.succ + 0 = k.succ := by
        simpa using congrArg Nat.succ h1
      exact hk
