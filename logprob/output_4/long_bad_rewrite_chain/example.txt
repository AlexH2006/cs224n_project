=== CONTEXT ===
import Mathlib

theorem long_bad_rewrite_chain (a b c : Nat) (hab : a = b) (hbc : b = c) : a + a = c + c := by

=== PROOF ===
  have h1 : a + a = b + a := by
    rw [hab]
  have h2 : b + a = b + b := by
    rw [hbc]
  have h3 : b + b = c + b := by
    rw [hbc]
  have h4 : c + b = c + c := by
    rw [hab]
  calc
    a + a = b + a := h1
    _ = b + b := h2
    _ = c + b := h3
    _ = c + c := h4
